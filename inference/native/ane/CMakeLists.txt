cmake_minimum_required(VERSION 3.20)
project(inkling_ane LANGUAGES CXX OBJCXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_OBJCXX_STANDARD 17)
find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
find_package(nanobind CONFIG REQUIRED)
find_package(MLX CONFIG REQUIRED)
nanobind_add_module(_inkling_ane NB_DOMAIN mlx ane.mm)
target_compile_options(_inkling_ane PRIVATE -fobjc-arc)
target_link_libraries(_inkling_ane PRIVATE mlx "-framework Foundation" "-framework CoreML")
